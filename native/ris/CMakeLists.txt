cmake_minimum_required(VERSION 3.16)
project(ris_native LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(ris STATIC src/ris.cpp)
target_include_directories(ris PUBLIC include)
target_compile_features(ris PUBLIC cxx_std_17)
target_link_libraries(ris PUBLIC Threads::Threads)
option(RIS_NATIVE "Tune for the build host CPU" OFF)
if(RIS_NATIVE)
    target_compile_options(ris PRIVATE -march=native)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(test_core tests/test_core.cpp)
    target_include_directories(test_core PRIVATE include)
    target_compile_features(test_core PRIVATE cxx_std_17)
    target_link_libraries(test_core PRIVATE Threads::Threads)
    add_test(NAME core COMMAND test_core)
endif()
